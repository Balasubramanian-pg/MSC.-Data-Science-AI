def clean_text(text: str) -> str:
    """
    Normalizes text by lowercasing, removing punctuation, and standardizing spaces.
    (In production, NLTK/Spacy is used for advanced stop-word removal and lemmatization).
    """
    text = text.lower()
    text = re.sub(r'[^\w\s]', '', text)  # Remove punctuation
    text = re.sub(r'\s+', ' ', text).strip() # Standardize whitespace
    return text
